# --- radcam AR1335 bring-up -------------------------------------------------
# Camera I2C: CAM0 connector -> i2c-6, CAM1 connector -> i2c-4.
# Both are enabled so the module is found in whichever port it is plugged into.
dtparam=i2c_csi_dsi0=on
dtparam=i2c_csi_dsi1=on

# Dosimeter (LTC2485) lives on the 40-pin header I2C, GPIO2/GPIO3 -> i2c-1.
dtparam=i2c_arm=on

# Sensor ENABLE is on connector GPIO1 (IO1), not IO0 as on stock Pi wiring.
# Move the camera regulators onto the pins this board actually uses:
#   CAM0 IO1 = RP1 line 35, CAM1 IO1 = RP1 line 48.
dtparam=cam0_reg_gpio=35
dtparam=cam1_reg_gpio=48

# AR1335 on both connectors, so the module works in whichever port it is in.
# The port with no camera simply fails to probe.
#
# extclk: the AR1335 has no internal oscillator and stays mute on I2C without a
# 24 MHz EXTCLK. The Pi's camera connector carries no clock, so this routes RP1
# GPCLK3 out of connector GPIO0 (unused on this board) to supply one. Harmless
# if the module turns out to have its own oscillator.
#
# If the module turns out to have no oscillator of its own, the Pi can supply
# EXTCLK on connector GPIO0 by adding `,extclk` to both lines below plus:
#     gpio=34=a1     # CAM0 IO0 -> GPCLK3
#     gpio=46=a0     # CAM1 IO0 -> GPCLK3
# Tried on 2026-08-10: the clock enables at 24 MHz but does not reach the pad,
# and the sensor stayed silent, so it is left off. See DEVELOPMENT_STATE.md.
# The camera lives on CAM1. The CAM0 instance costs nothing - it simply fails
# to probe - and keeps the system working if the module is ever moved.
#
# NOTE: do NOT add `i2c_slow` here. It was a bring-up experiment; at 10 kHz the
# ~309-register mode table takes over a second to write and the CSI front end
# times out waiting for frames.
dtoverlay=ar1335
dtoverlay=ar1335,cam0

# Illumination LEDs: TPS922051 dimmed by PWM0_CHAN2 on GPIO18.
# Brightness is capped to 10% in software by radcam/led.py.
dtoverlay=radcam-led

# Flight comms: ADM2582E RS422 transceiver on uart0, GPIO14 (TXD) / GPIO15 (RXD).
# This is separate from the Pi 5 debug UART, which stays on /dev/ttyAMA10.
dtparam=uart0=on
